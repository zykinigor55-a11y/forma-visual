package com.plux.assistant;

import android.content.ContentResolver;
import android.content.ContentValues;
import android.net.Uri;
import android.os.Build;
import android.os.Environment;
import android.provider.MediaStore;
import android.util.Base64;

import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;

import java.io.OutputStream;

/**
 * PLUX V31: saves user-requested generated pictures/MP4 through Android MediaStore.
 * No broad storage permissions. Android 10+ only; older devices use browser share.
 * The filename is sanitized; arbitrary filesystem paths are never accepted.
 */
@CapacitorPlugin(name = "PLUXMediaSaver")
public final class PLUXMediaSaver extends Plugin {
    @PluginMethod
    public void saveBase64(PluginCall call) {
        final String encoded = call.getString("data");
        final String mime = call.getString("mimeType");
        final String requestedName = call.getString("fileName");
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.Q) {
            call.reject("Для сохранения в галерею нужен Android 10 или новее.");
            return;
        }
        if (encoded == null || encoded.isEmpty() || encoded.length() > 50_000_000) {
            call.reject("Файл пустой либо слишком большой.");
            return;
        }
        final boolean video = "video/mp4".equals(mime);
        final boolean image = "image/png".equals(mime)
            || "image/jpeg".equals(mime)
            || "image/webp".equals(mime);
        if (!video && !image) {
            call.reject("Этот формат нельзя сохранить в галерею.");
            return;
        }
        final String ext = video ? ".mp4" : "image/png".equals(mime) ? ".png"
            : "image/webp".equals(mime) ? ".webp" : ".jpg";
        String name = requestedName == null ? "" : requestedName
            .replaceAll("[^a-zA-Z0-9_.-]", "_")
            .replaceAll("\\.{2,}", ".")
            .replaceAll("^\\.+", "");
        if (name.length() < 2) name = "PLUX_" + System.currentTimeMillis();
        if (name.length() > 100) name = name.substring(0, 100);
        if (!name.toLowerCase(java.util.Locale.ROOT).endsWith(ext)) name += ext;
        byte[] content;
        try { content = Base64.decode(encoded, Base64.DEFAULT); }
        catch (IllegalArgumentException exception) {
            call.reject("Повреждённый файл.");
            return;
        }
        if (content.length == 0 || content.length > 38_000_000) {
            call.reject("Файл превышает допустимый размер.");
            return;
        }
        Uri target = null;
        final ContentResolver resolver = getContext().getContentResolver();
        try {
            final ContentValues values = new ContentValues();
            values.put(MediaStore.MediaColumns.DISPLAY_NAME, name);
            values.put(MediaStore.MediaColumns.MIME_TYPE, mime);
            values.put(MediaStore.MediaColumns.RELATIVE_PATH,
                (video ? Environment.DIRECTORY_MOVIES : Environment.DIRECTORY_PICTURES) + "/PLUX");
            values.put(MediaStore.MediaColumns.IS_PENDING, 1);
            target = resolver.insert(video
                ? MediaStore.Video.Media.getContentUri(MediaStore.VOLUME_EXTERNAL_PRIMARY)
                : MediaStore.Images.Media.getContentUri(MediaStore.VOLUME_EXTERNAL_PRIMARY), values);
            if (target == null) throw new IllegalStateException("Нет доступа к памяти устройства");
            try (OutputStream out = resolver.openOutputStream(target, "w")) {
                if (out == null) throw new IllegalStateException("Не удалось открыть файл");
                out.write(content);
                out.flush();
            }
            ContentValues done = new ContentValues();
            done.put(MediaStore.MediaColumns.IS_PENDING, 0);
            resolver.update(target, done, null, null);
            final JSObject result = new JSObject();
            result.put("uri", target.toString());
            call.resolve(result);
        } catch (Exception error) {
            if (target != null) resolver.delete(target, null, null);
            call.reject("Не удалось сохранить медиафайл: " + error.getClass().getSimpleName());
        }
    }
}
